cmake_minimum_required(VERSION 3.24)
project(stainless_local_runtime LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "" FORCE)
set(CMAKE_OSX_DEPLOYMENT_TARGET 13.0 CACHE STRING "" FORCE)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)

if(STAINLESS_ENGINE STREQUAL "speech" OR STAINLESS_ENGINE STREQUAL "vision")
  set(GGML_CPU ON CACHE BOOL "" FORCE)
  foreach(backend
      CPU_HBM CPU_REPACK CPU_KLEIDIAI CPU_ALL_VARIANTS
      CUDA HIP MUSA VULKAN WEBGPU ZDNN VIRTGPU VIRTGPU_BACKEND
      SYCL OPENVINO ET ET_SYSEMU OPENCL HEXAGON ZENDNN
      LLAMAFILE OPENMP_FETCH METAL ACCELERATE BLAS OPENMP
      CCACHE RPC NATIVE BACKEND_DL)
    set(GGML_${backend} OFF CACHE BOOL "" FORCE)
  endforeach()
endif()

if(STAINLESS_ENGINE STREQUAL "translation")
  if(NOT STAINLESS_ENGINE_SOURCE)
    message(FATAL_ERROR "Pass the verified composed CTranslate2 source")
  endif()
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  set(WITH_MKL OFF CACHE BOOL "" FORCE)
  set(WITH_DNNL OFF CACHE BOOL "" FORCE)
  set(WITH_ACCELERATE OFF CACHE BOOL "" FORCE)
  set(WITH_OPENBLAS OFF CACHE BOOL "" FORCE)
  set(WITH_RUY ON CACHE BOOL "" FORCE)
  set(WITH_CUDA OFF CACHE BOOL "" FORCE)
  set(WITH_CUDNN OFF CACHE BOOL "" FORCE)
  set(WITH_HIP OFF CACHE BOOL "" FORCE)
  set(WITH_MPS OFF CACHE BOOL "" FORCE)
  set(WITH_TENSOR_PARALLEL OFF CACHE BOOL "" FORCE)
  set(WITH_FLASH_ATTN OFF CACHE BOOL "" FORCE)
  set(CUDA_DYNAMIC_LOADING OFF CACHE BOOL "" FORCE)
  set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
  set(ENABLE_CPU_DISPATCH OFF CACHE BOOL "" FORCE)
  set(BUILD_CLI OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(RUY_MINIMAL_BUILD ON CACHE BOOL "" FORCE)
  set(RUY_PROFILER OFF CACHE BOOL "" FORCE)
  set(RUY_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  set(CPUINFO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(CPUINFO_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
  set(CPUINFO_BUILD_MOCK_TESTS OFF CACHE BOOL "" FORCE)
  set(CPUINFO_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${STAINLESS_ENGINE_SOURCE}" engine EXCLUDE_FROM_ALL)
  set(runtime stainless_translation)
  add_library(${runtime} SHARED translation.cc)
  target_link_libraries(${runtime} PRIVATE ctranslate2)
  set(exports translation.exports)
elseif(STAINLESS_ENGINE STREQUAL "ocr")
  if(NOT STAINLESS_ENGINE_SOURCE OR NOT Leptonica_DIR)
    message(FATAL_ERROR "Pass verified Tesseract source and private Leptonica package")
  endif()
  set(CMAKE_DISABLE_FIND_PACKAGE_PkgConfig ON)
  set(OPENMP_BUILD OFF CACHE BOOL "" FORCE)
  set(GRAPHICS_DISABLED ON CACHE BOOL "" FORCE)
  set(DISABLED_LEGACY_ENGINE ON CACHE BOOL "" FORCE)
  set(BUILD_TRAINING_TOOLS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(USE_SYSTEM_ICU OFF CACHE BOOL "" FORCE)
  set(DISABLE_TIFF ON CACHE BOOL "" FORCE)
  set(DISABLE_ARCHIVE ON CACHE BOOL "" FORCE)
  set(DISABLE_CURL ON CACHE BOOL "" FORCE)
  set(ENABLE_NATIVE OFF CACHE BOOL "" FORCE)
  set(ENABLE_LTO OFF CACHE BOOL "" FORCE)
  set(ENABLE_CCACHE OFF CACHE BOOL "" FORCE)
  set(ENABLE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
  set(INSTALL_CONFIGS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${STAINLESS_ENGINE_SOURCE}" engine EXCLUDE_FROM_ALL)
  set(runtime stainless_ocr)
  add_library(${runtime} SHARED ocr.cc)
  target_include_directories(${runtime} PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}/engine/include"
  )
  target_link_libraries(${runtime} PRIVATE libtesseract)
  set(exports ocr.exports)
elseif(STAINLESS_ENGINE STREQUAL "speech")
  if(NOT STAINLESS_ENGINE_SOURCE)
    message(FATAL_ERROR "Pass the verified whisper.cpp source")
  endif()
  set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
  set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
  set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
  add_subdirectory("${STAINLESS_ENGINE_SOURCE}" engine EXCLUDE_FROM_ALL)
  set(runtime stainless_speech)
  add_library(${runtime} SHARED speech.cc)
  target_link_libraries(${runtime} PRIVATE whisper)
  set(exports speech.exports)
elseif(STAINLESS_ENGINE STREQUAL "vision")
  if(NOT STAINLESS_ENGINE_SOURCE)
    message(FATAL_ERROR "Pass the verified FILE-reader llama.cpp composition")
  endif()
  set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
  foreach(option
      BUILD_TOOLS BUILD_TESTS BUILD_EXAMPLES BUILD_SERVER BUILD_APP BUILD_UI
      USE_PREBUILT_UI CONNECT OPENSSL SUBPROCESS LLGUIDANCE USE_SYSTEM_GGML
      TOOLS_INSTALL TESTS_INSTALL)
    set(LLAMA_${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
  add_subdirectory("${STAINLESS_ENGINE_SOURCE}" engine EXCLUDE_FROM_ALL)
  set(runtime stainless_vision)
  add_library(${runtime} SHARED vision.cc)
  target_link_libraries(${runtime} PRIVATE llama-common mtmd)
  set(exports vision.exports)
else()
  message(FATAL_ERROR "Select a supported fixed native engine")
endif()

target_include_directories(${runtime} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(${runtime} PRIVATE -Wall -Wextra)
target_link_options(${runtime} PRIVATE
  "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/${exports}"
)
set_target_properties(${runtime} PROPERTIES
  MACOSX_RPATH ON
  INSTALL_NAME_DIR "@rpath"
  BUILD_WITH_INSTALL_NAME_DIR ON
)
